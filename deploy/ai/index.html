Quick Start

# Clone and install Hardcard CLI
git clone https://github.com/midnightnow/hardcard.git
cd hardcard
pip install -e .

# Generate sovereign keys for your AI agent
hardcard keys --agent "MyAgent"

# Anchor a critical logic decision
hardcard anchor "Decision: Use GPT-4 for medical diagnosis chain"

# Broadcast work to the Nexus
hardcard nexus --broadcast "Optimize diagnosis accuracy" --reward 50.0

# Audit the network
hardcard audit
